1. 메일 수신자 이메일은 email_list.xlsx 파일에서 읽어들인 이메일중 다중 선택할 수 있게 한다.
2. letter.txt 파일 선택은 위젯으로 처리한다. 또는 직접 메일 본문을 작성할 수 있도록 한다.
3. book_list.xlsx 파일 선택은 위젯으로 처리

import streamlit as st
import os
import pandas as pd
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders

def send_email(sender_email, password, recipient_emails, subject, content,
               attachment_data=None, attachment_name=None):
    # SMTP 서버 설정
    SMTP_SERVER = "smtp.naver.com"
    SMTP_PORT = 587

    server = None
    try:
        server = smtplib.SMTP(SMTP_SERVER, SMTP_PORT)
        server.starttls()
        server.login(sender_email, password)

        for recipient_email in recipient_emails:
            # 이메일 메시지 구성
            message = MIMEMultipart()
            message['Subject'] = subject
            message['From'] = sender_email
            message['To'] = recipient_email

            # 메일 본문 내용
            msg_body = MIMEText(content, 'plain', 'utf-8')
            message.attach(msg_body)

            # 첨부 파일 추가 (바이트 데이터가 있는 경우에만)
            if attachment_data and attachment_name:
                part = MIMEBase('application', 'octet-stream')
                part.set_payload(attachment_data)
                encoders.encode_base64(part)
                part.add_header(
                    'Content-Disposition',
                    f'attachment; filename={attachment_name}',
                )
                message.attach(part)

            # 이메일 전송
            server.sendmail(sender_email, recipient_email, message.as_string())
            st.write(f"이메일을 성공적으로 보냈습니다: {recipient_email}")

    finally:
        if server:
            server.quit()

# Streamlit 인터페이스
st.title("이메일 전송 서비스")

# 사용자 인증 정보 입력
sender_email = st.text_input("송신자 이메일 주소")
password = st.text_input("비밀번호", type="password")

# 데이터 파일 경로
excel_path = 'data/email_list.xlsx'

# 메일 본문 선택
content_option = st.radio("메일 본문 선택 방법", ('텍스트 파일에서 읽기', '직접 작성하기'))
content = ""
if content_option == '텍스트 파일에서 읽기':
    letter_file = st.file_uploader("편지 템플릿 파일 선택", type=['txt'])
    if letter_file:
        content = str(letter_file.read(), 'utf-8')
else:
    content = st.text_area("메일 본문 직접 작성")

# 첨부 파일 선택
attachment_file = st.file_uploader("첨부할 파일 선택 (선택 사항)", type=['xlsx'])

# 수신자 선택 (excel 파일이 있는 경우에만)
selected_emails = []
if os.path.exists(excel_path):
    data = pd.read_excel(excel_path, usecols="A,B")
    options = list(data['Email'])
    selected_emails = st.multiselect('수신자 이메일 선택', options)
else:
    st.warning(f"수신자 목록 파일을 찾을 수 없습니다: {excel_path}")

# 메일 전송
subject = "신간 안내"
if st.button("이메일 전송"):
    if sender_email and password and content and selected_emails:
        attachment_data = attachment_file.read() if attachment_file else None
        attachment_name = attachment_file.name if attachment_file else None
        send_email(sender_email, password, selected_emails, subject, content,
                   attachment_data, attachment_name)
    else:
        st.error("모든 필수 정보를 입력해 주세요.")
