################# 웹페이지 요약하기 ################### """ WebBaseLoader https://reference.langchain.com/python/langchain-community/document_loaders/web_base/WebBaseLoader?utm_source=chatgpt.com WebBaseLoader 를 사용하여 웹페이지에서 본문 텍스트를 불러오고, RecursiveCharacterTextSplitter 로 텍스트를 분할한 후, LangChain의 요약 체인을 사용하여 뉴스를 요약하는 Streamlit 앱. """ import streamlit as st import streamlit.components.v1 as components from urllib.parse import quote, urlparse, urlunparse from langchain_community.document_loaders import WebBaseLoader from langchain_text_splitters import RecursiveCharacterTextSplitter from langchain_openai import ChatOpenAI from langchain_core.prompts import PromptTemplate from langchain.chains.summarize import load_summarize_chain def encode_url(url): # 한글 등 비ASCII 문자가 포함된 URL을 퍼센트 인코딩 parsed = urlparse(url) return urlunparse(parsed._replace(path=quote(parsed.path, safe='/'))) st.set_page_config(page_title="웹페이지 요약기") st.title("📰 웹페이지 요약기") EXAMPLE_URLS = { "ChatGPT (Wikipedia EN)": "https://en.wikipedia.org/wiki/ChatGPT", "LangChain (Wikipedia EN)": "https://en.wikipedia.org/wiki/LangChain", "RAG (Wikipedia EN)": "https://en.wikipedia.org/wiki/Retrieval-augmented_generation", } if "article_url" not in st.session_state: st.session_state.article_url = "" st.markdown("**예제 URL 선택:**") cols = st.columns(len(EXAMPLE_URLS)) for col, (label, url) in zip(cols, EXAMPLE_URLS.items()): if col.button(label): st.session_state.article_url = url article_url = st.text_input("뉴스 기사 URL을 입력하세요", value=st.session_state.article_url) if article_url: components.iframe(article_url, height=400, scrolling=True) api_key = st.text_input("OpenAI API 키를 입력하세요", type="password") if st.button("요약 생성"): if not article_url or not api_key: st.error("URL과 API 키를 모두 입력해주세요.") else: try: # 웹페이지에서 본문 텍스트 불러오기 loader = WebBaseLoader(encode_url(article_url)) documents = loader.load() # 텍스트 분할 splitter = RecursiveCharacterTextSplitter(chunk_size=3000, chunk_overlap=100) split_docs = splitter.split_documents(documents) # LLM 모델 설정 llm = ChatOpenAI( api_key=api_key, model="gpt-6-luna", max_tokens=3000, request_timeout=120 ) # 요약용 프롬프트 (명령문으로 수정) map_prompt = PromptTemplate( template="다음 뉴스 기사 내용을 간결하게 요약하라:\n```{text}```" ) combine_prompt = PromptTemplate( template="다음은 뉴스 기사 조각들의 요약이다:\n```{text}```\n이 내용을 7~9문장으로 종합 요약하라." ) # 체인 구성 및 실행 chain = load_summarize_chain( llm, chain_type="map_reduce", map_prompt=map_prompt, combine_prompt=combine_prompt ) summary = chain.invoke(split_docs) st.success("요약 결과:") st.write(summary["output_text"]) except Exception as e: st.error(f"오류 발생: {e}")
