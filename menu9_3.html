마케팅 문구 스타일 변환
기존 광고 문구를 입력하고 스타일을 선택하면 AI가 새로운 문구로 리라이팅합니다. 
import streamlit as st
from openai import OpenAI


def main():
    st.set_page_config(page_title="마케팅 문구 스타일 변경기", layout="centered")
    st.title("🎯 마케팅 문구 스타일 변경기")

    st.markdown("원하는 스타일에 맞게 기존 광고 문구를 새롭게 바꿔드립니다.")

    # 입력 필드
    original_text = st.text_area("📢 기존 마케팅 문구를 입력하세요", height=120, value="이 스마트워치는 건강한 하루를 도와주는 최고의 선택입니다.")

    style = st.selectbox(
        "🎨 변경할 스타일을 선택하세요",
        ["젊은 층 대상", "프리미엄 고급", "SNS 바이럴용", "공손하고 신뢰감 있게", "재치 있고 유머러스하게"]
    )

    # 버튼 클릭 시 처리
    if st.button("🔁 스타일 변경"):
        if not original_text.strip():
            st.warning("마케팅 문구를 입력해 주세요.")
        else:
            with st.spinner("GPT가 문구를 리라이팅 중입니다..."):

                # 한국어 프롬프트: 입력과 출력이 모두 한국어이므로 한국어로 작성했습니다
                prompt = f"""
                아래 마케팅 문구를 '{style}' 스타일에 맞춰 다시 작성해줘.
                문장은 1~2문장으로 간결하고 자연스럽게 만들어줘.

                원문: {original_text}
                """

                try:
                    # api_key를 지정하지 않으면 환경변수 OPENAI_API_KEY 값을 자동으로 읽어옵니다
                    client = OpenAI()
                    response = client.responses.create(
                        model="gpt-6-luna",
                        instructions="당신은 광고 문구 스타일 변경 전문가입니다.",
                        input=prompt,
                    )
                    result = response.output_text.strip()
                    st.success("✅ 스타일 변경 완료!")
                    st.markdown(result)
                except Exception as e:
                    st.error(f"❌ 오류 발생: {e}")


if __name__ == '__main__':
    main()
