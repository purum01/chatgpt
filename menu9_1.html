FIELD_KEYS = ["name", "com_name", "strength", "tone_manner", "keyword", "value"]


def askGpt(prompt, apikey):
    client = OpenAI(api_key=apikey)
    response = client.responses.create(
        model="gpt-6-luna",
        instructions="당신은 광고 카피라이터 전문가입니다. 간결하고 임팩트 있는 광고 문구를 작성해 주세요.",
        input=prompt
    )
    return response.output_text


def main():
    st.set_page_config(page_title="광고 문구 생성 프로그램")

    # session state 초기화
    if "OPENAI_API" not in st.session_state:
        st.session_state["OPENAI_API"] = ""
    for key in FIELD_KEYS:
        if f"field_{key}" not in st.session_state:
            st.session_state[f"field_{key}"] = EXAMPLES[0][key]

    # 사이드바
    with st.sidebar:
        open_apikey = st.text_input(label='OPENAI API 키', placeholder='Enter Your API Key', value='', type='password')
        if open_apikey:
            st.session_state["OPENAI_API"] = open_apikey
        st.markdown('---')

    # 메인공간
    st.header("🎸광고 문구 생성 프로그램")
    st.markdown('---')

    # 예제 버튼
    st.caption("예제 불러오기")
    cols = st.columns(len(EXAMPLES))
    for col, example in zip(cols, EXAMPLES):
        with col:
            if st.button(example["label"], use_container_width=True):
                for key in FIELD_KEYS:
                    st.session_state[f"field_{key}"] = example[key]
                st.rerun()
    st.markdown('---')

    col1, col2 = st.columns(2)
    with col1:
        name = st.text_input("제품명", key="field_name")
        strength = st.text_input("제품 특징", key="field_strength")
        keyword = st.text_input("필수 포함 키워드", key="field_keyword")
    with col2:
        com_name = st.text_input("브랜드 명", key="field_com_name")
        tone_manner = st.text_input("톤엔 메너", key="field_tone_manner")
        value = st.text_input("브랜드 핵심 가치", key="field_value")

    if st.button("광고 문구 생성"):
        if not st.session_state["OPENAI_API"]:
            st.warning("사이드바에 OpenAI API 키를 입력해 주세요.")
        else:
            prompt = f'''
            아래 내용을 참고해서 1~2줄짜리 광고문구 5개 작성해줘
            - 제품명: {name}
            - 브랜드 명: {com_name}
            - 브랜드 핵심 가치: {value}
            - 제품 특징: {strength}
            - 톤엔 매너: {tone_manner}
            - 필수 포함 키워드: {keyword}
            '''
            with st.spinner("광고 문구 생성 중..."):
                st.info(askGpt(prompt, st.session_state["OPENAI_API"]))


if __name__ == '__main__':
    main()
