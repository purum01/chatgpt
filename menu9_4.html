고객 문의 자동 분석
고객 문의를 입력하면 AI가 문의 요약·유형·긴급도·감정·추천 대응·답변 초안을 자동으로 분석합니다. 
################# AI 고객 문의 분석기 ###################

import streamlit as st
from openai import OpenAI


def askGpt(customer_message, apikey):

    client = OpenAI(api_key=apikey)

    response = client.responses.create(
        model="gpt-6-luna",
        instructions="""당신은 전문적인 고객 지원 담당자입니다.

고객 문의를 분석하여 다음 형식으로 답변하세요.

[문의 요약]
고객 문의의 핵심 내용을 한 문장으로 요약

[문의 유형]
배송 / 환불 / 교환 / 제품 / 결제 / 기타 중 하나

[긴급도]
높음 / 보통 / 낮음

[고객 감정]
고객의 현재 감정을 분석

[추천 대응]
상담원이 취해야 할 조치

[답변 초안]
고객에게 보낼 친절하고 전문적인 답변""",
        input=customer_message
    )

    return response.output_text


# 초기화 함수 — main() 바깥에 있어도 st.session_state는 전역으로 공유되므로 접근 가능합니다
def reset():
    st.session_state["customer_message"] = ""
    st.session_state["result"] = ""


def main():

    st.set_page_config(page_title="AI 고객 문의 분석기")

    # session state 초기화
    if "OPENAI_API" not in st.session_state:
        st.session_state["OPENAI_API"] = ""

    if "customer_message" not in st.session_state:
        st.session_state["customer_message"] = ""

    if "result" not in st.session_state:
        st.session_state["result"] = ""

    # 사이드바
    with st.sidebar:

        open_apikey = st.text_input(
            label="OPENAI API 키",
            placeholder="Enter Your API Key",
            value="",
            type="password"
        )

        if open_apikey:
            st.session_state["OPENAI_API"] = open_apikey

        st.markdown("---")

    # 메인 화면
    st.header("🤖 AI 고객 문의 분석기")

    st.write(
        "고객 문의를 입력하면 AI가 문의 내용과 "
        "고객 감정을 분석하고 대응 방법을 제안합니다."
    )

    st.markdown("---")

    # 고객 문의
    customer_message = st.text_area(
        "고객 문의를 입력하세요",
        height=250,
        placeholder="""예)
주문한 노트북이 예정일보다 5일이나 늦었는데
아직도 배송되지 않았습니다.
이번 주까지 필요해서 주문한 건데 너무 실망스럽네요.
        """,
        key="customer_message"
    )

    # 버튼
    col1, col2 = st.columns(2)

    with col1:
        analyze = st.button(
            "🔍 AI 분석",
            type="primary",
            use_container_width=True
        )

    with col2:
        st.button(
            "🔄 초기화",
            use_container_width=True,
            on_click=reset
        )

    # AI 분석
    if analyze:

        if not st.session_state["OPENAI_API"]:
            st.warning("OPENAI API 키를 입력하세요.")

        elif not customer_message:
            st.warning("고객 문의를 입력하세요.")

        else:

            with st.spinner("AI가 고객 문의를 분석하고 있습니다..."):

                result = askGpt(
                    customer_message,
                    st.session_state["OPENAI_API"]
                )

                # 분석 결과 저장
                st.session_state["result"] = result

    # 분석 결과 출력
    if st.session_state["result"]:

        st.markdown("---")

        st.subheader("📌 AI 분석 결과")

        st.markdown(st.session_state["result"])


if __name__ == "__main__":
    main()
