################# 회의록 자동 요약하기 ################### import streamlit as st from openai import OpenAI def askGpt(prompt, apikey): client = OpenAI(api_key=apikey) response = client.responses.create( model="gpt-6-luna", instructions="당신은 회의록 요약 전문가입니다. 핵심 내용을 간결하고 명확하게 정리해 주세요.", input=prompt ) return response.output_text def main(): st.set_page_config(page_title="회의록 요약") # session state 초기화 if "OPENAI_API" not in st.session_state: st.session_state["OPENAI_API"] = "" # 사이드바 with st.sidebar: open_apikey = st.text_input(label='OPENAI API 키', placeholder='Enter Your API Key', value='', type='password') if open_apikey: st.session_state["OPENAI_API"] = open_apikey st.markdown('---') st.header("📃회의록 요약 프로그램") st.markdown('---') transcript = st.text_area("요약 할 회의록을 입력하세요", height=300) if st.button("요약"): if not st.session_state["OPENAI_API"]: st.warning("사이드바에 OpenAI API 키를 입력해 주세요.") elif not transcript.strip(): st.warning("회의록 내용을 입력해 주세요.") else: prompt = f"다음 회의록을 요약해 주세요:\n\n{transcript}" with st.spinner("요약 중..."): st.info(askGpt(prompt, st.session_state["OPENAI_API"])) if __name__ == "__main__": main()
