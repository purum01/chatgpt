""" PDF를 기반으로 답변하는 챗봇 서비스 pip install streamlit langchain langchain-community langchain-openai openai faiss-cpu tiktoken pypdf """ import os import tempfile import streamlit as st from langchain_community.document_loaders import PyPDFLoader from langchain_openai import ChatOpenAI, OpenAIEmbeddings from langchain_text_splitters import CharacterTextSplitter from langchain_community.vectorstores import FAISS from langchain.chains import RetrievalQA # 카테고리별 예시 질문 EXAMPLE_QUESTIONS = { "단순 검색": [ "RFA 시술 후 임신한 환자는 총 몇 명인가?", "연구 대상자의 평균 연령은 얼마인가?", "연구 기간은 언제부터 언제까지인가?", "RFA 시술 전 자궁근종의 중앙값 부피는 얼마인가?", "임신까지 걸린 중앙값 기간은 얼마인가?", "자연임신 환자는 몇 명인가?", "ART(보조생식술)로 임신한 환자는 몇 명인가?", "유산율은 몇 %인가?", "제왕절개 비율은 몇 %인가?", "조산율은 몇 %인가?", "자간전증 발생률은 몇 %인가?", ], "표 검색": [ "RFA 후 6개월 시점의 근종 부피 감소율은 얼마인가?", "RFA 후 12개월 시점의 근종 부피 감소율은 얼마인가?", "평균 수술 시간은 얼마인가?", "평균 퇴원 시간은 얼마인가?", "정상 생활로 복귀하는 데 걸린 시간은 얼마인가?", "Type IIIa 합병증은 몇 명에게 발생했는가?", "ART군과 자연임신군의 평균 연령 차이는 얼마인가?", "ART군과 자연임신군의 유산율 차이는 얼마인가?", ], "내용 정리 (여러 chunk 종합)": [ "RFA 시술 후 임신 결과를 요약해줘.", "RFA 시술 후 산모와 태아의 안전성에 대해 설명해줘.", "연구진은 RFA 후 임신 시기를 어떻게 권고하고 있는가?", "연구 결과를 바탕으로 RFA가 가임력 보존 치료로 적절한 이유를 설명해줘.", "RFA 시술의 장점은 무엇인가?", ], "추론 검색": [ "근종 크기와 임신까지 걸리는 시간은 어떤 관계가 있는가?", "유산율에 영향을 주는 요인은 무엇이었는가?", "왜 연구진은 RFA 후 최소 6개월 이후 임신을 권장하는가?", "ART군이 자연임신군보다 유산율이 높았던 이유는 무엇인가?", "근종 개수와 임신까지 걸리는 시간은 관련이 있었는가?", ], "RAG 성능 확인": [ "연구 결과 자궁 파열(uterine rupture)은 발생했는가?", "연구 결과 태반 조기 박리(placental abruption)는 발생했는가?", "태아 성장 제한(fetal growth restriction)은 발생했는가?", "가장 심각한 합병증은 무엇이었는가?", "태반 유착(placenta accreta)이 발생한 사례는 어떤 환자였는가?", "연구진이 언급한 연구의 한계점은 무엇인가?", "RFA 후 임신 결과를 일반 인구와 비교하면 어떠한가?", ], } def build_knowledge_base(pdf_file, api_key): with tempfile.NamedTemporaryFile(delete=False, suffix=".pdf") as tmp: tmp.write(pdf_file.read()) tmp_path = tmp.name try: loader = PyPDFLoader(tmp_path) documents = loader.load() finally: os.remove(tmp_path) text_splitter = CharacterTextSplitter( separator="\n", chunk_size=1000, chunk_overlap=200, length_function=len, ) chunks = text_splitter.split_documents(documents) embeddings = OpenAIEmbeddings(api_key=api_key) return FAISS.from_documents(chunks, embeddings) def set_question(q): st.session_state["pending_question"] = q ##### 메인 함수 ##### def main(): st.set_page_config(page_title="PDF analyzer", layout="wide") if "pending_question" not in st.session_state: st.session_state["pending_question"] = "" # 사이드바 with st.sidebar: default_key = (os.getenv("OPENAI_API_KEY") or "").strip() open_apikey = st.text_input("OpenAI API Key", value=default_key, type="password") if open_apikey: st.session_state["OPENAI_API"] = open_apikey st.markdown('---') # 메인공간 st.header("PDF 내용 질문 프로그램📜") st.markdown('---') st.subheader("PDF 파일을 넣으세요") pdf = st.file_uploader(" ", type="pdf") if pdf is not None: api_key = st.session_state.get("OPENAI_API", "") if st.session_state.get("pdf_name") != pdf.name: with st.spinner("PDF 인덱스 생성 중..."): st.session_state["knowledge_base"] = build_knowledge_base(pdf, api_key) st.session_state["pdf_name"] = pdf.name st.success("PDF 로드 완료!") st.markdown('---') # 예시 질문 (카테고리별) st.subheader("📋 예시 질문 (클릭하면 자동 입력)") for category, questions in EXAMPLE_QUESTIONS.items(): with st.expander(category): cols = st.columns(2) for i, q in enumerate(questions): cols[i % 2].button( q, key=f"eq_{category}_{i}", on_click=set_question, args=(q,), use_container_width=True, ) st.markdown('---') st.subheader("질문을 입력하세요") user_question = st.text_input( "Ask a question about your PDF:", value=st.session_state["pending_question"], ) if st.button("질문하기"): if not api_key: st.error("API 키를 입력하세요.") elif not user_question.strip(): st.warning("질문을 입력하세요.") else: with st.spinner("답변 생성 중..."): llm = ChatOpenAI( api_key=api_key, max_tokens=3000, model='gpt-6-luna', request_timeout=120, ) qa_chain = RetrievalQA.from_chain_type( llm=llm, chain_type="stuff", retriever=st.session_state["knowledge_base"].as_retriever(), ) response = qa_chain.invoke({"query": user_question}) st.session_state["pending_question"] = "" st.info(response['result']) if __name__ == '__main__': main()
