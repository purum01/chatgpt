################# 유튜브 동영상 요약하기 ################### """ YouTubeTranscriptApi https://github.com/jdepoix/youtube-transcript-api YouTube 동영상의 자막(Transcript)을 YouTubeTranscriptApi로 불러와 RecursiveCharacterTextSplitter 로 텍스트를 분할한 후, LangChain의 요약 체인을 사용하여 동영상 내용을 요약하는 예제. """ import os from youtube_transcript_api import YouTubeTranscriptApi from langchain_core.documents import Document from langchain_text_splitters import RecursiveCharacterTextSplitter from langchain_openai import ChatOpenAI from langchain_core.prompts import PromptTemplate from langchain.chains.summarize import load_summarize_chain video_id = "Pn-W41hC764" transcript_list = YouTubeTranscriptApi.get_transcript(video_id, languages=['en']) text = " ".join([item['text'] for item in transcript_list]) transcript = [Document(page_content=text, metadata={"source": video_id})] text_splitter = RecursiveCharacterTextSplitter(chunk_size=4000) text = text_splitter.split_documents(transcript) api_key = os.getenv('OPENAI_API_KEY') llm = ChatOpenAI(api_key=api_key, max_tokens=3000, model="gpt-6-luna", request_timeout=120) # 각각의 chunk를 요약하기 prompt = PromptTemplate( template="Summarize the youtube video whose transcript is provided within backticks ```{text}```" ) # 요약된 내용들을 취합하여 다시 한번 요약하기 combine_prompt = PromptTemplate( template="""Combine all the youtube video summaries provided within backticks ```{text}``` Provide a concise summary between 8 to 10 sentences.""" ) chain = load_summarize_chain(llm, chain_type="map_reduce", verbose=False, map_prompt=prompt, combine_prompt=combine_prompt) result = chain.invoke(text) print(result['output_text'])
